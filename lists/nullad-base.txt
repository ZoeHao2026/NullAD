! Section 4: option-restricted rules

! Only block ad frames, so a site that legitimately embeds content is unaffected.
||doubleclick.net^$subdocument
||googlesyndication.com^$subdocument

! Block ad scripts but never navigate away from a page.
||adnxs.com^$script
||adsrvr.org^$script
||pubmatic.com^$script
||rubiconproject.com^$script

! Block tracking pixels and beacons only.
||scorecardresearch.com^$image,ping
||quantserve.com^$image,ping
||omtrdc.net^$image,ping
||2o7.net^$image,ping

! Everything except scripts: used by lists that only object to passive tracking.
||demdex.net^$~script
||bluekai.com^$~script

! ---------------------------------------------------------------------------
! Section 5: domain-constrained rules
! $domain= restricts a rule to specific initiators, which is how a list avoids
! breaking the one site that depends on an otherwise-unwanted host.
! ---------------------------------------------------------------------------
||google-analytics.com^$domain=~example.com
||hotjar.com^$domain=~example.com
||criteo.com^$domain=~example.com
